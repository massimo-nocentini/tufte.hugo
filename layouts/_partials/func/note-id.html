{{- /* Returns a new note id, unique within the rendered HTML of a page. Every producer of margin-toggle markup (sidenotes, margin notes, margin figures, figure captions and footnotes turned into sidenotes) calls this partial, so the ids never collide, not even on list pages that show the summaries of several pages. The id is "-", where n is a per-page counter kept in .Store under the key "tufteNoteCounter". Callers prefix it with "sn-" (sidenote) or "mn-" (margin note). Takes: a page. Returns: the id string, e.g. "3f2a9c-4". */ -}} {{- $page := . -}} {{- /* Store.Add increments under one lock (atomic read-modify-write): pages render in parallel, and a page's notes can be produced by its own page and by a list showing its summary at the same time. The counter only grows, so each rendering gets strictly increasing ids. */ -}} {{- $page.Store.Add "tufteNoteCounter" 1 -}} {{- $n := int ($page.Store.Get "tufteNoteCounter") -}} {{- $hash := substr (md5 (printf "%s" (or $page.RelPermalink $page.Path))) 0 6 -}} {{- return (printf "%s-%d" $hash $n) -}}
