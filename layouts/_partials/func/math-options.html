{{- /* KaTeX settings for a page, from [params.math] and front matter mathMacros. Takes a dict: Page the page required Display bool: options for display math (server engine) Returns a dict: engine "server" (default) | "client" font "katex" (default) | "euler": params.math.font, or the page's front matter mathFont. "euler" makes KaTeX output MathML only (server.output "mathml", client.output "mathml"), which the browser lays out with the bundled Euler Math font (README "Euler math") server options for transform.ToMath (output htmlAndMathml, displayMode, throwOnError false, errorColor, fleqn, leqno, minRuleThickness, strict, macros) client the same options for KaTeX in the browser (no output, no displayMode) errorColor params.math.errorColor as set by the site ("" if unset) sentinel the errorColor KaTeX actually gets (both engines): KaTeX colours an undefined control sequence (a typo, a missing macro) with errorColor and no katex-error class, so render-passthrough and tufte/katex.html look for this colour to catch it Macros: params.math.macros, then the page's mathMacros (which win). Both accept two forms: a map macros = { "\\half" = "\\frac{1}{2}" } a list of pairs macros = [ ["\\RR", "\\mathbb{R}"], ["\\abs", "\\left|#1\\right|"] ] Hugo lowercases the keys of maps in the configuration and in front matter, so a map entry "\\RR" becomes "\\rr": use the list of pairs (or a list of {name = "…", value = "…"} tables) for any macro name with capital letters. */ -}} {{- $page := .Page -}} {{- $display := eq .Display true -}} {{- $cfg := partialCached "func/math-config.html" "math" "math" -}} {{- $engine := lower (printf "%v" (index $cfg "engine" | default "server")) -}} {{- if not (in (slice "server" "client") $engine) -}} {{- warnidf "tufte-bad-option" "tufte.hugo: params.math.engine must be \"server\" or \"client\", not %q; using \"server\"" $engine -}} {{- $engine = "server" -}} {{- end -}} {{- /* Macros: a Scratch map keeps the keys exactly as written (merge and site params are case-insensitive). */ -}} {{- $s := newScratch -}} {{- range $src := slice (index $cfg "macros") (index $page.Params "mathmacros") -}} {{- if reflect.IsMap $src -}} {{- range $k, $v := $src -}} {{- $s.SetInMap "macros" (printf "%v" $k) (printf "%v" $v) -}} {{- end -}} {{- else if reflect.IsSlice $src -}} {{- range $src -}} {{- if and (reflect.IsSlice .) (eq (len .) 2) -}} {{- $s.SetInMap "macros" (printf "%v" (index . 0)) (printf "%v" (index . 1)) -}} {{- else if and (reflect.IsMap .) (index . "name") -}} {{- $s.SetInMap "macros" (printf "%v" (index . "name")) (printf "%v" (index . "value" | default "")) -}} {{- else -}} {{- warnidf "tufte-bad-option" "tufte.hugo: ignoring math macro %v (expected [\"\\\\name\", \"expansion\"]) in %s" . $page.Path -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $macros := $s.Get "macros" | default (dict) -}} {{- $errorColor := printf "%v" (index $cfg "errorcolor" | default "") -}} {{- /* func/flag.html: a HUGO_PARAMS_MATH_FLEQN=true string too. */ -}} {{- $fleqn := partial "func/flag.html" (dict "value" (index $cfg "fleqn") "default" false) -}} {{- $leqno := partial "func/flag.html" (dict "value" (index $cfg "leqno") "default" false) -}} {{- $minRule := 0.04 -}} {{- with index $cfg "minrulethickness" -}} {{- /* A number of em: 0.05, "0.05" (an environment variable) or "0.05em". Anything else would stop the build in float. */ -}} {{- $v := strings.TrimSpace (printf "%v" .) | strings.TrimSuffix "em" | strings.TrimSpace -}} {{- if findRE `^[0-9]*\.?[0-9]+([eE][-+]?[0-9]+)?$` $v 1 -}} {{- $minRule = float $v -}} {{- else -}} {{- warnidf "tufte-bad-option" "tufte.hugo: params.math.minRuleThickness must be a number (in em), not %q; using 0.04" (printf "%v" .) -}} {{- end -}} {{- end -}} {{- $strict := lower (printf "%v" (index $cfg "strict" | default "error")) -}} {{- if not (in (slice "error" "ignore" "warn") $strict) -}} {{- warnidf "tufte-bad-option" "tufte.hugo: params.math.strict must be \"error\", \"ignore\" or \"warn\", not %q; using \"error\"" $strict -}} {{- $strict = "error" -}} {{- end -}} {{- /* The math font: the page's mathFont, else params.math.font. */ -}} {{- $font := "katex" -}} {{- range $src := slice (index $cfg "font") (index $page.Params "mathfont") -}} {{- with $src -}} {{- $v := lower (strings.TrimSpace (printf "%v" .)) -}} {{- if in (slice "katex" "euler") $v -}} {{- $font = $v -}} {{- else if $v -}} {{- warnidf "tufte-bad-option" "tufte.hugo: the math font (params.math.font or front matter mathFont) must be \"katex\" or \"euler\", not %q, in %s; ignored" $v $page.Path -}} {{- end -}} {{- end -}} {{- end -}} {{- $output := cond (eq $font "euler") "mathml" "htmlAndMathml" -}} {{- /* Distinctive (and still red, should it ever be shown) colour. */ -}} {{- $sentinel := "#cc0001" -}} {{- /* The dict is built in one go (merge would lowercase the macro names). */ -}} {{- $server := dict "output" $output "displayMode" $display "throwOnError" false "errorColor" $sentinel "fleqn" $fleqn "leqno" $leqno "minRuleThickness" $minRule "strict" $strict "macros" $macros -}} {{- /* Client: the sentinel too; tufte/katex.html replaces it with the .katex-error class (the theme's colour, adapted to dark mode) plus the site's errorColor, if any, inline. */ -}} {{- $client := dict "throwOnError" false "errorColor" $sentinel "fleqn" $fleqn "leqno" $leqno "minRuleThickness" $minRule "strict" $strict "macros" $macros -}} {{- if eq $font "euler" -}} {{- $client = dict "output" "mathml" "throwOnError" false "errorColor" $sentinel "fleqn" $fleqn "leqno" $leqno "minRuleThickness" $minRule "strict" $strict "macros" $macros -}} {{- end -}} {{- return (dict "engine" $engine "font" $font "server" $server "client" $client "errorColor" $errorColor "sentinel" $sentinel) -}}
