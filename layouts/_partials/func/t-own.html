{{- /* Which UI strings a language has of its own, for func/t.html: Hugo's T falls back to the defaultContentLanguage's string, which func/t.html must not show in another language (README, Languages: English is the fallback). A theme language (the five i18n/*.yaml files, every key in each) has them all. Another language has the keys of the site's own files i18n/.yaml (.yml, .toml or .json) whose base name is the language key in any casing, as Hugo reads them (pt-br, pt-BR, zh-Hans, zh-Hant-TW, sr-Latn), listed and read relative to the project directory (os.ReadDir, os.ReadFile), so that the result does not depend on the filesystem being case-insensitive. Keys compare lowercased, as Hugo's T. Takes: the language key (string). Called through partialCached. Returns: dict all bool: the language has every key (a theme language) keys map lowercased key → true (when not all) */ -}} {{- $name := lower (printf "%v" .) -}} {{- $out := dict "all" false "keys" (dict) -}} {{- if in (slice "en" "it" "fr" "de" "es") $name -}} {{- $out = dict "all" true "keys" (dict) -}} {{- else -}} {{- $files := slice -}} {{- with try (os.ReadDir "i18n") -}} {{- range .Value -}} {{- if not .IsDir -}} {{- $ext := strings.TrimPrefix "." (path.Ext .Name) -}} {{- if and (in (slice "yaml" "yml" "toml" "json") $ext) (eq (lower (strings.TrimSuffix (path.Ext .Name) .Name)) $name) -}} {{- $files = $files | append (dict "f" (printf "i18n/%s" .Name) "ext" $ext) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $keys := dict -}} {{- range $files -}} {{- $ext := .ext -}} {{- with try (os.ReadFile .f | transform.Unmarshal (dict "format" (cond (eq $ext "yml") "yaml" $ext))) -}} {{- with .Value -}} {{- if reflect.IsMap . -}} {{- range $k, $v := . }}{{ $keys = merge $keys (dict (lower $k) true) }}{{ end -}} {{- else -}} {{- /* go-i18n v1 list: - id: key translation: … */ -}} {{- range . -}} {{- if reflect.IsMap . }}{{ with .id }}{{ $keys = merge $keys (dict (lower (printf "%v" .)) true) }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $out = dict "all" false "keys" $keys -}} {{- end -}} {{- return $out -}}
