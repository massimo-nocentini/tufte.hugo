{{- /* Emits the single theme stylesheet. In order: ET Book @font-face rules (unless Params.etBook is false or "false", func/flag.html), tufte.css, theme.css, tufte-latex.css, syntax.css and every path listed in Params.customCSS (resolved in the site's assets/; their @import statements go first). The bundle is published as css/tufte-hugo..css, so the relative font URLs in et-book.css (../et-book/...) resolve against static/et-book/. Minified in production builds only, always fingerprinted with SRI. */ -}} {{- $paths := slice -}} {{- if partial "func/flag.html" (dict "value" site.Params.etBook "default" true) -}} {{- $paths = $paths | append "css/et-book.css" -}} {{- end -}} {{- $paths = $paths | append "css/tufte.css" -}} {{- $paths = $paths | append "css/theme.css" -}} {{- /* The tufte-latex layer: class options, numbered floats, book views, print geometry (its page-dependent values: tufte/head-style.html). */ -}} {{- $paths = $paths | append "css/tufte-latex.css" -}} {{- $paths = $paths | append "css/syntax.css" -}} {{- $resources := slice -}} {{- range $paths -}} {{- with resources.Get . -}} {{- $resources = $resources | append . -}} {{- else -}} {{- errorf "tufte.hugo: theme stylesheet %q not found in assets/" . -}} {{- end -}} {{- end -}} {{- /* customCSS: a list of paths; a single string (`customCSS = "css/custom.css"`, or HUGO_PARAMS_CUSTOMCSS, which is always a string) is a list of comma-separated paths. Anything else is reported and ignored. */ -}} {{- $custom := slice -}} {{- with site.Params.customCSS -}} {{- if reflect.IsSlice . -}} {{- $custom = . -}} {{- else if eq (printf "%T" .) "string" -}} {{- range split . "," }}{{ with strings.TrimSpace . }}{{ $custom = $custom | append . }}{{ end }}{{ end -}} {{- else -}} {{- warnidf "tufte-bad-option" "tufte.hugo: customCSS must be a list of paths under assets/ (or one path), not %v; ignored" . -}} {{- end -}} {{- end -}} {{- /* A browser ignores an @import that follows other rules, and the custom files come last in the bundle: their @import statements (a web font stylesheet, say) are moved to the top of the bundle, in order, and removed from the files. */ -}} {{- $imports := slice -}} {{- /* The URL may hold ";" (Google Fonts: wght@400;700), so a quoted or url(…) URL is matched whole, then the media list up to ";". */ -}} {{- $importRE := `(?i)@import\s*(?:url\(\s*(?:"[^"]*"|'[^']*'|[^)]*)\s*\)|"[^"]*"|'[^']*')[^;]*;` -}} {{- range $custom -}} {{- with resources.Get (printf "%v" .) -}} {{- $r := . -}} {{- with findRE $importRE $r.Content -}} {{- $imports = $imports | append . -}} {{- $r = resources.FromString (printf "tufte/custom-css/%s" $r.Name) (replaceRE $importRE "" $r.Content) -}} {{- end -}} {{- $resources = $resources | append $r -}} {{- else -}} {{- warnidf "tufte-bad-option" "tufte.hugo: customCSS %q not found in assets/, skipped" . -}} {{- end -}} {{- end -}} {{- with $imports -}} {{- $resources = slice (resources.FromString "tufte/custom-css/imports.css" (delimit . "\n")) | append $resources -}} {{- end -}} {{- with $resources -}} {{- $css := . | resources.Concat "css/tufte-hugo.css" -}} {{- if hugo.IsProduction -}} {{- $css = $css | minify -}} {{- end -}} {{- $css = $css | fingerprint }} {{- end -}}
