{{- /* One line of page metadata: date (and update date), author, reading time, word count, categories, tags. Rendered only for pages that have a date, categories or tags (or author sidenotes to show). Params: showReadingTime (default true), showWordCount (default false); both can be overridden in front matter. Front matter, as in the latex title block (\date): `showDate: false` hides the date, `dateText` replaces it with free text ("Spring 2026"). Authors: tufte/authors.html (a string, a list of strings or a list of {name, thanks}; names joined "A, B and C"). Takes: the page, or a dict with page the page authorHTML optional: the author HTML to show (tufte/title-block.html passes the names with their thanks sidenotes) authorNotes optional: how many sidenotes authorHTML holds */ -}} {{- $p := . -}} {{- $author := "" -}} {{- $notes := 0 -}} {{- if reflect.IsMap . -}} {{- $p = .page -}} {{- with .authorHTML }}{{ $author = . }}{{ end -}} {{- with .authorNotes }}{{ $notes = int . }}{{ end -}} {{- end -}} {{- if not $author -}} {{- $author = (partial "tufte/authors.html" (dict "page" $p)).html -}} {{- end -}} {{- $tags := $p.GetTerms "tags" -}} {{- $categories := $p.GetTerms "categories" -}} {{- $showDate := ne $p.Params.showdate false -}} {{- $dateText := "" -}} {{- with $p.Params.datetext }}{{ $dateText = printf "%v" . }}{{ end -}} {{- if or (not $p.Date.IsZero) $tags $categories (and $showDate $dateText) (gt $notes 0) -}} {{- $sep := false }}

{{- if $showDate -}} {{- if $dateText -}} {{ T "postedOn" }} {{ $dateText }} {{- $sep = true -}} {{- else if not $p.Date.IsZero -}} {{ T "postedOn" }} {{ partial "func/date.html" $p.Date }} {{- if and (not $p.Lastmod.IsZero) (gt ($p.Lastmod.Format "2006-01-02") ($p.Date.Format "2006-01-02")) }} ({{ T "updatedOn" }} {{ partial "func/date.html" $p.Lastmod }}) {{- end -}} {{- $sep = true -}} {{- end -}} {{- end -}} {{- with $author -}} {{- if $sep }} · {{ end -}} {{ T "by" }} {{ . | safeHTML }} {{- $sep = true -}} {{- end -}} {{- if and (ne ($p.Param "showReadingTime") false) (gt $p.ReadingTime 0) -}} {{- if $sep }} · {{ end -}} {{ T "readingTime" $p.ReadingTime }} {{- $sep = true -}} {{- end -}} {{- if and ($p.Param "showWordCount") (gt $p.WordCount 0) -}} {{- if $sep }} · {{ end -}} {{ T "wordCount" $p.WordCount }} {{- $sep = true -}} {{- end -}} {{- with $categories -}} {{- if $sep }} · {{ end -}} {{- range $i, $c := . -}} {{- if $i }}, {{ end -}} {{ $c.LinkTitle }} {{- end -}} {{- $sep = true -}} {{- end -}} {{- with $tags -}} {{- if $sep }} · {{ end -}} {{- range $i, $t := . -}} {{- if $i }} {{ end -}} {{ $t.LinkTitle }} {{- end -}} {{- end -}}

{{- end -}}
