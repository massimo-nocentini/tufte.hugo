{{- /* The bibliography database seen by a page (SPEC-LATEX §5.1). Takes: a page. Returns: a map lowercased cite key → raw entry (native map or CSL item). Callers cache it: partialCached "tufte/bib/db.html" $page (or $page.RelPermalink $page.Path) Sources, later ones winning on the same key: 1. the site database (hugo.Data at option bibliography.data); 2. in a book, the book-wide pool: the `bibliography` and `references` of every part and chapter (tufte/bib/book-pool.html; the first page in reading order wins on a key), so a key defined by one chapter can be cited from any page of the book, as LaTeX has one database per document; 3. in a book, the book root's front matter `bibliography` (file) and `references` (inline entries); 4. the page's front matter `bibliography`: a path (or a list of paths) to a page resource, else a global asset (assets/), in .json, .yaml, .yml, .toml (decoded with transform.Unmarshal) or .bib (bibtex-lite); 5. the page's front matter `references`: a map key → entry, or a list of CSL items. Inside a book (any page but the root), sources 4 and 5 do not override a key that 2 or 3 hold: the book has one database, and as BibTeX keeps the first of repeated entries everywhere, a chapter that redefines a key of the root or of an earlier chapter cites the book's entry, the one the book's bibliography lists. A redefinition with a different entry warns (tufte-duplicate-cite), once, from this page. This partial never reads page content, so cite shortcodes may call it while the page is rendering. Sources 4 and 5 come from tufte/bib/source-db.html, cached per source page (and per file text), so a book's .bib file is parsed once per build. A chapter without sources of its own (the usual case), or whose sources the book pool already holds, shares its book root's database instead of copying it. */ -}} {{- /* Memoized in the page's Store for the current build (tufte/memo.html): partialCached's bounded cache evicts it on large books, and every chapter reads it. */ -}} {{- $memoRes := false -}} {{- with partial "tufte/memo.html" (dict "page" . "key" "tufteMemoBibDb") -}} {{- $memoRes = .v -}} {{- else -}} {{- $p := . -}} {{- $dk := partialCached "tufte/opt.html" (dict "page" $p "key" "bibliography.data" "default" "bibliography") (or $p.RelPermalink $p.Path) "bibliography.data" -}} {{- $dk = printf "%v" ($dk | default "bibliography") -}} {{- $own := partialCached "tufte/bib/source-db.html" $p (or $p.RelPermalink $p.Path) -}} {{- /* the base: the book root's database (site data + book pool + root sources) when it reads the same site data, else the site data, the book pool and the root sources */ -}} {{- $base := false -}} {{- $bookKeys := dict -}} {{- with partialCached "tufte/book/root.html" $p (or $p.RelPermalink $p.Path) -}} {{- $pool := partialCached "tufte/bib/book-pool.html" . (or .RelPermalink .Path) -}} {{- $isRoot := eq (or .RelPermalink .Path) (or $p.RelPermalink $p.Path) -}} {{- if and $own (not $isRoot) -}} {{- /* the keys the book itself holds (pool and root sources): a chapter's own sources cannot override them */ -}} {{- $K := newScratch -}} {{- range $k, $_ := $pool }}{{ $K.Set $k true }}{{ end -}} {{- range $k, $_ := partialCached "tufte/bib/source-db.html" . (or .RelPermalink .Path) }}{{ $K.Set $k true }}{{ end -}} {{- $bookKeys = $K.Values -}} {{- end -}} {{- $rdk := $dk -}} {{- if not $isRoot -}} {{- $rdk = partialCached "tufte/opt.html" (dict "page" . "key" "bibliography.data" "default" "bibliography") (or .RelPermalink .Path) "bibliography.data" -}} {{- $rdk = printf "%v" ($rdk | default "bibliography") -}} {{- end -}} {{- if and (not $isRoot) (eq $rdk $dk) -}} {{- $base = partialCached "tufte/bib/db.html" . (or .RelPermalink .Path) -}} {{- else -}} {{- /* the root itself (its own sources are added below), or a page that reads other site data than its root */ -}} {{- $R := newScratch -}} {{- $R.Set "db" dict -}} {{- range $k, $v := partialCached "tufte/bib/site-db.html" $dk $dk }}{{ $R.SetInMap "db" $k $v }}{{ end -}} {{- range $k, $v := $pool }}{{ $R.SetInMap "db" $k $v }}{{ end -}} {{- if not $isRoot -}} {{- range $k, $v := partialCached "tufte/bib/source-db.html" . (or .RelPermalink .Path) }}{{ $R.SetInMap "db" $k $v }}{{ end -}} {{- end -}} {{- $base = $R.Get "db" -}} {{- end -}} {{- end -}} {{- if not $base }}{{ $base = partialCached "tufte/bib/site-db.html" $dk $dk }}{{ end -}} {{- $out := $base -}} {{- /* the page's own entries, less the keys of its book */ -}} {{- $A := newScratch -}} {{- range $k, $v := $own -}} {{- if not (isset $bookKeys $k) -}} {{- $A.Set $k $v -}} {{- else if ne (jsonify $v) (jsonify (index $base $k)) -}} {{- /* uncached: normalize.html's cache key (page, key) means the entry this page resolves, not its overridden one */ -}} {{- $mine := partial "tufte/bib/normalize.html" (dict "entry" $v "key" $k) -}} {{- $book := partial "tufte/bib/normalize.html" (dict "entry" (index $base $k) "key" $k) -}} {{- if ne (jsonify $mine) (jsonify $book) -}} {{- $where := or $p.RelPermalink $p.Path -}} {{- with $p.File }}{{ $where = .Path }}{{ end -}} {{- warnidf "tufte-duplicate-cite" "tufte.hugo: %s: cite key %q redefines an entry of the book (its root or an earlier chapter) with a different one; a book has one database, so the first definition is kept, as BibTeX does (\"Repeated entry\")" $where $k -}} {{- end -}} {{- end -}} {{- end -}} {{- with $A.Values -}} {{- $S := newScratch -}} {{- $S.Set "db" dict -}} {{- range $k, $v := $base }}{{ $S.SetInMap "db" $k $v }}{{ end -}} {{- range $k, $v := . }}{{ $S.SetInMap "db" $k $v }}{{ end -}} {{- $out = $S.Get "db" -}} {{- end -}} {{- $memoRes = $out -}} {{- .Store.Set "tufteMemoBibDb" (dict "v" $memoRes "b" (partialCached "tufte/build-stamp.html" "stamp" "stamp")) -}} {{- end -}} {{- return $memoRes -}}
