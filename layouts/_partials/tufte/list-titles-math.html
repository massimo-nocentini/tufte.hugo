{{- /* Whether a page shows another page's title, in the list mode of tufte/title-html.html (math typeset), that holds math delimiters ($, \(, \[), so that head.html loads KaTeX for it (math.enable = "auto"). The titles are those of: - the page's categories and tags (post-meta.html, the taxonomy line of the LaTeX title block); - home and section pages outside a book: the pages of the current pager (post-summary.html) and their categories and tags; - the archive page (layout "archive"): the posts of the main sections (archive.html); - a term page: its pages (tufte/archive-list.html) and its taxonomy (the link back); - a taxonomy page: its terms (taxonomy.html). The page's own title is checked by head.html (front matter). Takes: the page. Returns: bool. */ -}} {{- $re := `\$|\\\(|\\\[` -}} {{- $titles := slice -}} {{- $terms := slice "categories" "tags" -}} {{- range $terms }}{{ range $.GetTerms . }}{{ $titles = $titles | append .LinkTitle }}{{ end }}{{ end -}} {{- $inBook := partialCached "tufte/book/root.html" . (or .RelPermalink .Path) -}} {{- if and (not $inBook) (or .IsHome (eq .Kind "section")) -}} {{- range (partial "func/paginate.html" .).Pages -}} {{- $titles = $titles | append .LinkTitle -}} {{- $q := . -}} {{- range $terms }}{{ range $q.GetTerms . }}{{ $titles = $titles | append .LinkTitle }}{{ end }}{{ end -}} {{- end -}} {{- else if eq .Layout "archive" -}} {{- $main := site.Params.mainSections | default site.MainSections -}} {{- range where site.RegularPages "Type" "in" $main }}{{ $titles = $titles | append .LinkTitle }}{{ end -}} {{- else if eq .Kind "term" -}} {{- range .Pages }}{{ $titles = $titles | append .LinkTitle }}{{ end -}} {{- with .Parent }}{{ $titles = $titles | append .Title .LinkTitle }}{{ end -}} {{- else if eq .Kind "taxonomy" -}} {{- range .Data.Terms.Alphabetical }}{{ $titles = $titles | append .Page.LinkTitle }}{{ end -}} {{- end -}} {{- return (gt (len (findRE $re (delimit $titles "\n") 1)) 0) -}}
