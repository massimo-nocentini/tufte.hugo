{{- /* Running counter sums of a whole book, in one pass (for tufte/offsets.html). Walks the print order of the book once (the same walk as list.print.html: the landing, then tufte/book/sequence.html: the chapters in reading order, each part page right before the first chapter of the part, parts without chapters at their place) and records, for every page, the sums of tufte/counts.html over the pages before it. Computing this per page would walk the whole sequence for every chapter (quadratic in the number of chapters). Takes: the book root page. Returns: map page key → dict sidenote figure table listing eq (ints); page key = printf "%s|%s" .Path (or .RelPermalink ""). Callers: partialCached "tufte/book/offset-map.html" $root (or $root.RelPermalink $root.Path) */ -}} {{- /* Memoized in the root's Store (not in hugo server): see tufte/opts.html. */ -}} {{- $memoRes := dict -}} {{- with cond hugo.IsServer nil (.Store.Get "tufteMemoOffsetMap") -}} {{- $memoRes = .v -}} {{- else -}} {{- $root := . -}} {{- $seq := slice $root -}} {{- range partialCached "tufte/book/sequence.html" $root (or $root.RelPermalink $root.Path) -}} {{- $seq = $seq | append .Page -}} {{- end -}} {{- $sn := 0 -}}{{- $fig := 0 -}}{{- $tab := 0 -}}{{- $lst := 0 -}}{{- $eq := 0 -}} {{- $M := newScratch -}} {{- range $seq -}} {{- $M.SetInMap "m" (printf "%s|%s" .Path (or .RelPermalink "")) (dict "sidenote" $sn "figure" $fig "table" $tab "listing" $lst "eq" $eq) -}} {{- $c := partialCached "tufte/counts.html" . (or .RelPermalink .Path) -}} {{- $sn = add $sn $c.sidenote -}} {{- $fig = add $fig $c.figure -}} {{- $tab = add $tab $c.table -}} {{- $lst = add $lst $c.listing -}} {{- $eq = add $eq $c.eq -}} {{- end -}} {{- $memoRes = $M.Get "m" | default dict -}} {{- if not hugo.IsServer }}{{ $root.Store.Set "tufteMemoOffsetMap" (dict "v" $memoRes) }}{{ end -}} {{- end -}} {{- return $memoRes -}}
