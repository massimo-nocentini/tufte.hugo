{{- /* Book-wide label map (SPEC-LATEX §6.4, R11): the merge of tufte/labels.html entries over the book root, its parts and its chapters (reading order). Labels are global within a book: an xref that misses on its own page looks here. Each entry carries its page. The first definition wins; the same explicit label in two pages gives one tufte-duplicate-label warning (this partial is cached, so it is warned once per build). Same ranks as tufte/labels.html: automatic heading ids (they repeat across chapters: "introduction", …) < \label anchors < the rest, explicit heading ids ({#sec:x}) included; a higher rank wins silently, automatic heading ids never conflict. Takes: the book root page. Returns: map normalised key → entry dict (see tufte/labels.html). Callers: partialCached "tufte/book/labels.html" $root (or $root.RelPermalink $root.Path) */ -}} {{- /* Memoized in the root's Store for the current build: see tufte/opts.html. Every chapter's xrefs read it. */ -}} {{- $memoRes := false -}} {{- with partial "tufte/memo.html" (dict "page" . "key" "tufteMemoBookLabels") -}} {{- $memoRes = .v -}} {{- else -}} {{- $root := . -}} {{- $pages := slice $root -}} {{- range partialCached "tufte/book/parts.html" $root (or $root.RelPermalink $root.Path) -}} {{- $pages = $pages | append .Page -}} {{- end -}} {{- range partialCached "tufte/book/chapters.html" $root (or $root.RelPermalink $root.Path) -}} {{- $pages = $pages | append .Page -}} {{- end -}} {{- $ranks := dict "heading" 0 "label" 1 -}} {{- $E := newScratch -}} {{- $E.Set "e" dict -}} {{- $dups := slice -}} {{- range $pages -}} {{- $L := partialCached "tufte/labels.html" . (or .RelPermalink .Path) -}} {{- range $k, $v := $L.entries -}} {{- /* Markdown footnote labels are page-local (tufte/labels.html). */ -}} {{- if eq $v.local true }}{{ continue }}{{ end -}} {{- $old := index ($E.Get "e") $k -}} {{- /* An explicit heading id ({#sec:x}) is an author label: rank 2, as in tufte/labels.html, so the same one in two chapters is warned; automatic ids (they repeat: "introduction") stay 0. */ -}} {{- $rank := 2 -}} {{- if and (isset $ranks $v.kind) (not (and (eq $v.kind "heading") (eq $v.explicit true))) }}{{ $rank = index $ranks $v.kind }}{{ end -}} {{- $oldRank := 0 -}} {{- with $old -}} {{- $oldRank = 2 -}} {{- if and (isset $ranks .kind) (not (and (eq .kind "heading") (eq .explicit true))) }}{{ $oldRank = index $ranks .kind }}{{ end -}} {{- end -}} {{- if not $old -}} {{- $E.SetInMap "e" $k $v -}} {{- else if gt $rank $oldRank -}} {{- $E.SetInMap "e" $k $v -}} {{- else if and (eq $rank $oldRank) (gt $rank 0) (ne $old.page $v.page) -}} {{- $at := $v.page.RelPermalink -}} {{- with $v.page.File }}{{ $at = .Path }}{{ end -}} {{- $dups = $dups | append (printf "%s (%s)" $k $at) -}} {{- end -}} {{- end -}} {{- end -}} {{- with $dups -}} {{- $at := $root.RelPermalink -}} {{- with $root.File }}{{ $at = .Path }}{{ end -}} {{- warnidf "tufte-duplicate-label" "tufte.hugo: duplicate label(s) across the book %s: %s (the first one in reading order wins)" $at (delimit . ", ") -}} {{- end -}} {{- $memoRes = $E.Get "e" -}} {{- $root.Store.Set "tufteMemoBookLabels" (dict "v" $memoRes "b" (partialCached "tufte/build-stamp.html" "stamp" "stamp")) -}} {{- end -}} {{- return $memoRes -}}
