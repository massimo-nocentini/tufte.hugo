{{- /* The class list of : the LaTeX document class and the class options that are implemented in CSS (assets/css/tufte-latex.css). tufte-book | tufte-handout the page is inside a book (§4) or not tufte-print-book the whole-book print document (list.print.html) tufte-style-latex tufte.style = "latex" tufte-justified tufte.justified tufte-hyphenate tufte.hyphenate tufte-indent tufte.indent tufte-sf tufte.sfsidenotes tufte-nols tufte.letterspacing = false tufte-osf tufte.numerals = "oldstyle" tufte-nocite-numbers tufte.bibliography.margin = false tufte-debug tufte.debug Every option is read with tufte/opt.html (page → part → book → site → style preset), so a book's `tufte:` front matter reaches its chapters. Takes: the page. Returns: the class string. */ -}} {{- $p := . -}} {{- $classes := slice -}} {{- $root := partialCached "tufte/book/root.html" $p (or $p.RelPermalink $p.Path) -}} {{- if $root -}} {{- $classes = $classes | append "tufte-book" -}} {{- else -}} {{- $classes = $classes | append "tufte-handout" -}} {{- end -}} {{- /* The print output format: Hugo leaves the format being rendered out of
