{{- /* A grid of images with one caption (block; subfigures, \fbox grids). Put it on its own lines, the images on CONSECUTIVE lines (one paragraph, so they are inline images; images separated by blank lines would each become a figure of their own): {{< subfigures cols=2 type="fullwidth" caption="Title pages." label="fig:titles" >}} ![Handout](title-handout.png "Handout") ![Book](title-book.png "Book") {{< /subfigures >}} Each image title becomes a subcaption "(a) Handout", "(b) Book": inline Markdown, like any caption ("*Book*", "$x^2$"), taken off the . Output: */ -}} {{- $p := .Page -}} {{- $get := dict -}} {{- if .IsNamedParams }}{{ $get = .Params }}{{ end -}} {{- $cols := 2 -}} {{- with index $get "cols" -}} {{- $v := printf "%v" . | strings.TrimSpace | replaceRE `^([0-9]+)\.0+$` "$1" -}} {{- $n := 0 -}} {{- if findRE `^[0-9]{1,3}$` $v 1 }}{{ $n = int (strings.TrimLeft "0" $v | default "0") }}{{ end -}} {{- if and (ge $n 1) (le $n 12) }}{{ $cols = $n }}{{ else }}{{ warnidf "tufte-bad-option" "subfigures shortcode: cols must be 1–12, not %q (%s)" $v $.Position }}{{ end -}} {{- end -}} {{- $html := partial "tufte/render-inner.html" (dict "page" $p "inner" (strings.TrimSpace .Inner) "display" "inline" "where" (printf "%s shortcode in %v" .Name .Position)) | string -}} {{- $imgs := findRE `(?:]*>)?]*>(?:)?` $html -}} {{- if not $imgs -}} {{- warnidf "tufte-empty" "subfigures shortcode: no images found; write them as ![alt](src \"title\") on consecutive lines (%s)" .Position -}} {{- end -}} {{- $letters := split "abcdefghijklmnopqrstuvwxyz" "" -}} {{- $cells := "" -}} {{- range $i, $img := $imgs -}} {{- $sub := "" -}} {{- with findRESubmatch `\stitle="([^"]*)"` $img 1 -}} {{- /* The title is inline Markdown, like every caption ($x$, *em*); it becomes the subcaption, so it leaves the img (no tooltip repeating it). */ -}} {{- $t := $p.RenderString (dict "display" "inline") (htmlUnescape (index . 0 1)) | string | strings.TrimSpace -}} {{- $sub = printf `(%s) %s` (index $letters (mod $i 26)) $t -}} {{- $img = replaceRE `\stitle="[^"]*"` "" $img 1 -}} {{- end -}} {{- $cells = printf `%s%s%s` $cells $img $sub -}} {{- end -}} {{- $body := printf `
%s
` $cells -}} {{- $out := partial "tufte/figure.html" (dict "Page" $p "Body" $body "Kind" "figure" "Type" (index $get "type" | default "main") "Style" (printf "--cols:%d" $cols) "Caption" (index $get "caption" | default "") "Class" (printf "figure-grid %v" (index $get "class" | default "") | strings.TrimSpace) "Label" (index $get "label" | default "") "Short" (index $get "short" | default "") "Numbered" (index $get "numbered") "CaptionPos" (index $get "captionpos" | default "") "Offset" (index $get "offset" | default "") "Cite" (index $get "cite" | default "") ) -}} {{- partial "tufte/nest.html" (dict "ctx" . "html" $out) | safeHTML -}}
