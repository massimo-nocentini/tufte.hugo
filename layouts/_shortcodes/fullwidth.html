{{- /* Full-width block (block): lets its content use the whole article width. Put it on its own lines; the inner text is block Markdown (a table, a code block, paragraphs, …): {{< fullwidth >}} | A | B | |---|---| | 1 | 2 | {{< /fullwidth >}} Params: class (optional, extra classes). Output (Tufte CSS):
…
*/ -}} {{- $text := strings.TrimSpace .Inner -}} {{- /* Block Markdown; nested shortcodes are kept (nest tokens). */ -}} {{- $html := partial "tufte/render-inner.html" (dict "page" .Page "inner" $text "text" $text "display" "block" "where" (printf "%s shortcode in %v" .Name .Position)) | string -}} {{- $class := "" -}} {{- if .IsNamedParams -}} {{- $class = printf "%v" (.Get "class" | default "") | strings.TrimSpace -}} {{- end -}} {{- $out := printf "
\n%s\n
" (cond (ne $class "") (printf " %s" (htmlEscape $class)) "") $html -}} {{- partial "tufte/nest.html" (dict "ctx" . "html" $out) | safeHTML -}}
