>}} Params (named): src video file: page resource, assets/ resource, static path or URL (required) poster image shown before playback (same resolution as src) caption caption, inline Markdown: a margin note (main) or a figcaption (fullwidth) title accessible label of the video (default: the caption as plain text) type "main" (default, main text column) | "fullwidth" controls default true; "src" "poster" -}} {{- $v := printf "%v" ($.Get $key | default "") | strings.TrimSpace -}} {{- $u := $v -}} {{- $remote := or (strings.HasPrefix $v "http://") (strings.HasPrefix $v "https://") (strings.HasPrefix $v "//") (strings.HasPrefix $v "data:") -}} {{- if and $v (not $remote) -}} {{- $path := strings.TrimPrefix "./" $v -}} {{- $res := "" -}} {{- with $p.Resources.Get $path -}} {{- $res = . -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $path) -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- with $res -}} {{- $u = .RelPermalink -}} {{- if eq $key "src" }}{{ $mime = .MediaType.Type }}{{ end -}} {{- else -}} {{- if strings.HasPrefix $v "/" -}} {{- $u = strings.TrimPrefix "/" $v | relURL -}} {{- end -}} {{- end -}} {{- end -}} {{- $urls = merge $urls (dict "%v" (.Get "type" | default "main")) -}} {{- if not (in (slice "main" "fullwidth") $type) -}} {{- warnf "video shortcode: unknown type %q, using \"main\" (%s)" $type .Position -}} {{- $type = "main" -}} {{- end -}} {{- /* Caption: inline Markdown, never block elements (it may sit in a span). */ -}} {{- $caption := "" -}} {{- with .Get "caption" -}} {{- $html := partial "tufte/render-inner.html"
