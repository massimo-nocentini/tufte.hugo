{{- /* Self-hosted video (block), as a Tufte CSS figure. Put it on its own line: {{< video src="flow.mp4" poster="flow.jpg" caption="Water flow, *slowed down*." >}} {{< video src="/media/flow.webm" type="fullwidth" loop="true" muted="true" autoplay="true" >}} Params (named): src video file: page resource, assets/ resource, static path or URL (required) poster image shown before playback (same resolution as src) A data: URI is accepted for an image poster and a video or audio src; other non-http(s) schemes become "#ZgotmplZ". caption caption, inline Markdown: a margin note (main) or a figcaption.caption-margin (fullwidth) title accessible label of the video (default: the caption as plain text) type "main" (default, main text column) | "fullwidth" controls default true; "false" hides the player controls autoplay, loop, muted, playsinline "true" to set (autoplay implies muted and playsinline, as browsers require) width, height video attributes class extra classes for the figure Output (Tufte CSS): (caption-margin, as a full-width image's: in the note column under the video on wide screens, and in the margin column in print.) */ -}} {{- if not .IsNamedParams -}} {{- errorf "video shortcode: use named parameters, e.g. src=\"clip.mp4\" caption=\"…\" (%s)" .Position -}} {{- end -}} {{- $p := .Page -}} {{- $yes := slice "true" "1" "yes" -}} {{- /* Resolve src and poster: page resource, assets/ resource, then URL. */ -}} {{- $urls := dict -}} {{- $mime := "" -}} {{- range $key := slice "src" "poster" -}} {{- $v := printf "%v" ($.Get $key | default "") | strings.TrimSpace -}} {{- $u := $v -}} {{- $remote := or (strings.HasPrefix $v "http://") (strings.HasPrefix $v "https://") (strings.HasPrefix $v "//") (strings.HasPrefix $v "data:") -}} {{- if and $v (not $remote) -}} {{- /* tufte/resolve-res.html: also "my%20clip.mp4", "clip.mp4#t=5". */ -}} {{- $found := partial "tufte/resolve-res.html" (dict "page" $p "src" $v) -}} {{- $res := $found.res -}} {{- with $res -}} {{- $u = $found.url -}} {{- if eq $key "src" }}{{ $mime = .MediaType.Type }}{{ end -}} {{- else -}} {{- if strings.HasPrefix $v "/" -}} {{- $u = strings.TrimPrefix "/" $v | relURL -}} {{- end -}} {{- end -}} {{- end -}} {{- $urls = merge $urls (dict $key $u) -}} {{- end -}} {{- $src := index $urls "src" -}} {{- if not $src -}} {{- errorf "video shortcode: the src parameter is required (%s)" .Position -}} {{- end -}} {{- $type := lower (printf "%v" (.Get "type" | default "main")) -}} {{- if not (in (slice "main" "fullwidth") $type) -}} {{- warnidf "tufte-bad-option" "video shortcode: unknown type %q, using \"main\" (%s)" $type .Position -}} {{- $type = "main" -}} {{- end -}} {{- /* Caption: inline Markdown, never block elements (it may sit in a span). As for the caption= of a figure (tufte/figure.html): shortcodes in it run; a cite becomes part of one citation block after the caption, and a sidenote or margin note stays inline text (tufte/note-in-note.html), since the caption is itself a margin note. */ -}} {{- $where := printf "%s shortcode in %v" .Name .Position -}} {{- $caption := "" -}} {{- with .Get "caption" -}} {{- $html := partial "tufte/render-inner.html" (dict "page" $p "inner" (printf "%v" .) "display" "inline" "shortcodes" "run" "where" $where) | string -}} {{- $html = partial "func/flatten-p.html" $html | strings.TrimSpace -}} {{- if strings.Contains $html `class="margin-toggle` -}} {{- $html = partial "tufte/note-in-note.html" (dict "page" $p "html" $html "holder" "video caption" "where" $where) -}} {{- end -}} {{- with partial "tufte/cite-gather.html" (dict "page" $p "html" $html) -}} {{- $html = printf "%s%s" (.html | strings.TrimSpace) .block -}} {{- end -}} {{- $caption = $html -}} {{- end -}} {{- /* The default accessible label: the caption's plain text, without its notes, citations, toggles and KaTeX's MathML copy (as a figure's alt text); cross-references are resolved by tufte/postprocess.html. */ -}} {{- $label := printf "%v" (.Get "title" | default "") -}} {{- if and (not $label) $caption -}} {{- $label = partial "tufte/plain-caption.html" $caption -}} {{- end -}} {{- $controls := not (in (slice "false" "0" "no") (printf "%v" (.Get "controls"))) -}} {{- $autoplay := in $yes (printf "%v" (.Get "autoplay")) -}} {{- $loop := in $yes (printf "%v" (.Get "loop")) -}} {{- $muted := or $autoplay (in $yes (printf "%v" (.Get "muted"))) -}} {{- $inline := or $autoplay (in $yes (printf "%v" (.Get "playsinline"))) -}} {{- $width := printf "%v" (.Get "width" | default "") -}} {{- $height := printf "%v" (.Get "height" | default "") -}} {{- $class := printf "%v" (.Get "class" | default "") | strings.TrimSpace -}} {{- $figClass := "video" -}} {{- if eq $type "fullwidth" }}{{ $figClass = printf "%s fullwidth" $figClass }}{{ end -}} {{- with $class }}{{ $figClass = printf "%s %s" $figClass . }}{{ end -}} {{- $id := "" -}} {{- if and (eq $type "main") $caption -}} {{- $id = partial "func/note-id.html" $p -}} {{- end -}} {{- /* Built as a string (attributes escaped as html/template would, script URLs refused) so that it goes through tufte/nest.html: nested in a theme shortcode (fullwidth, marginblock…) its HTML is kept, also with markup.goldmark.renderer.unsafe = false. A data: URI is kept when it holds media that is never executed: an image for poster, a video or audio for src (whose MIME type then becomes the type); any other URL goes through tufte/safe-url.html (http, https, mailto or no scheme). */ -}} {{- $safe := dict -}} {{- $dataSrc := false -}} {{- range $key := slice "src" "poster" -}} {{- $raw := printf "%v" (index $urls $key | default "") | replaceRE `[\t\n\r]` "" | replaceRE `^[\x00-\x20]+|[\x00-\x20]+$` "" -}} {{- $kinds := cond (eq $key "poster") "image" "video|audio" -}} {{- $u := "" -}} {{- with findRESubmatch (printf `^(?i:data):((?i:%s)/[a-zA-Z0-9!#$&^_.+-]+)[;,]` $kinds) $raw 1 -}} {{- $u = $raw -}} {{- if and (eq $key "src") (not $mime) }}{{ $mime = lower (index . 0 1) }}{{ end -}} {{- if eq $key "src" }}{{ $dataSrc = true }}{{ end -}} {{- else -}} {{- $u = partial "tufte/safe-url.html" $raw -}} {{- end -}} {{- $safe = merge $safe (dict $key $u) -}} {{- end -}} {{- $out := printf `` $out -}} {{- partial "tufte/nest.html" (dict "ctx" . "html" $out) | safeHTML -}}
