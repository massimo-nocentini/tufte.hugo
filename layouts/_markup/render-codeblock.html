{{- /* Code block render hook. - Fenced code with a language is highlighted as usual with Chroma (transform.HighlightCodeBlock, honouring the [markup.highlight] settings and the info-string options); languages Chroma does not know are rendered as plain text. - Fenced code without a language renders as Tufte CSS's 
.
  - A fullwidth class widens the block:

      ```clojure {.fullwidth}
      (map tufte-css blog-posts)
      ```

    → 
…

      (Hugo copies the block attributes onto the highlight wrapper; if it
      did not, the block is wrapped in 
 instead)
      (without a language: 
…
)

  Mermaid diagrams are not supported.
*/ -}}
{{- $fullwidth := findRE `(^|\s)fullwidth(\s|$)` (printf "%v" (index .Attributes "class" | default "")) 1 -}}
{{- if .Type -}}
  {{- $opts := dict -}}
  {{- if not (transform.CanHighlight .Type) -}}
    {{- $opts = dict "type" "text" -}}
  {{- end -}}
  {{- $result := transform.HighlightCodeBlock . $opts -}}
  {{- if and $fullwidth (not (findRE `^\s*]*\sclass="[^"]*\bfullwidth\b` (string $result.Wrapped) 1)) -}}

{{ $result.Wrapped }}

  {{- else -}}
    {{- $result.Wrapped -}}
  {{- end -}}
{{- else -}}
{{ .Inner }}
{{- end -}}
