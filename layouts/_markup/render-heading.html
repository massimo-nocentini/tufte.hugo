{{- /* Heading render hook: keeps the heading text as rendered and adds a small anchor link, shown on hover by theme.css:

Figures#

Tufte CSS styles h1 (title), h2 (sections) and h3; deeper levels are rendered but not specially styled. */ -}} {{- $extra := "" -}} {{- range $k, $v := .Attributes -}} {{- $val := printf "%v" $v -}} {{- if and $val (ne $k "id") (findRE `^[a-zA-Z][a-zA-Z0-9_:.-]*$` $k 1) (not (strings.HasPrefix (lower $k) "on")) -}} {{- $extra = printf `%s %s="%s"` $extra $k (htmlEscape $val) -}} {{- end -}} {{- end -}} {{ .Text }} {{- with .Anchor }}#{{ end -}} {{- /* no trailing newline */ -}}
