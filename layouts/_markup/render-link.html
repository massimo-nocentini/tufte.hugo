{{- /* Link render hook. Tufte CSS styles plain elements, so the markup is unchanged; destinations are resolved like Hugo's embedded link hook: a relative destination that names a page (e.g. "../other-post.md" or "/about") or a page/global resource is replaced by its relative permalink, keeping any query and fragment. Fragment-only links ("#x"), absolute URLs, dot-only paths ("./", "../") and unresolved relative paths are left as written, for the browser to resolve against the page (in a summary on a list page, tufte/postprocess.html points "#x" to the page: PAGE#x). An unresolved root-absolute path ("/images/x.svg", "/index.xml") is resolved against the site's base path (tufte/site-url.html), like an image src, so that it also works on a site hosted under a subpath. Notes in the link text ([text[^1]](url), a sidenote, marginnote or cite shortcode) are moved after the by tufte/postprocess.html (tufte/unnest-links.html), and links nested in it are unlinked. A linked stand-alone image, [![alt](src "Caption")](url), reaches the image hook with .IsBlock = true and becomes a figure, which may not sit inside an (nor inside the

Goldmark keeps around the link). The link then goes inside the figure, around the only, and the figure is wrapped in inert block markers:

func/content-raw.html drops the paragraph around a block that is alone in it (the figure stands on its own, like an unlinked block image), and turns a block that shares its paragraph with text into a linked plain image followed by its caption (func/inline-figures.html, which also handles blocks in notes). */ -}} {{- $href := .Destination -}} {{- $u := "" -}} {{- with try (urls.Parse .Destination) -}} {{- if not .Err -}}{{- $u = .Value -}}{{- end -}} {{- end -}} {{- $safe := true -}} {{- with $u -}} {{- if in (slice "javascript" "vbscript" "data" "file") (lower .Scheme) -}} {{- $safe = false -}} {{- end -}} {{- /* A path of dot segments only ("./", ".", "../", "../..") names no page: GetPage "" is the home page and a relative "../" resolves against the content file, not the URL. It is left as written, so the browser resolves it against the page (and a summary elsewhere gets the same URL from tufte/postprocess.html). */ -}} {{- if and .Path (not .IsAbs) (not .Host) (not (findRE `^\.\.?(?:/\.\.?)*/?$` .Path 1)) -}} {{- $path := strings.TrimPrefix "./" .Path -}} {{- $target := "" -}} {{- with $.PageInner.GetPage $path -}} {{- $target = .RelPermalink -}} {{- else -}} {{- with $.PageInner.Resources.Get $path -}} {{- $target = .RelPermalink -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $path) -}} {{- $target = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- /* An unresolved root-absolute path ("/images/x.svg", "/index.xml") names a file under the site root: it is resolved against the site's base path like an image src (tufte/site-url.html), so it also works on a site hosted under a subpath. */ -}} {{- if and (not $target) (strings.HasPrefix .Path "/") (not (strings.HasPrefix .Path "//")) -}} {{- $target = partial "tufte/site-url.html" .EscapedPath -}} {{- end -}} {{- with $target -}} {{- $href = . -}} {{- with $u.RawQuery -}}{{- $href = printf "%s?%s" $href . -}}{{- end -}} {{- with $u.Fragment -}}{{- $href = printf "%s#%s" $href . -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $safe = false -}} {{- end -}} {{- /* A shortcode (or a nest token) in the title: Hugo would put its HTML inside the attribute. The title is deferred like an image's alt text (tufte/attr-marks.html): a token in the attribute, the escaped title after the ; its plain text becomes the tooltip and its notes and citations follow the link once the shortcodes have run. */ -}} {{- /* The title as written, with its backslash escapes and entities resolved like Goldmark would (tufte/md-title.html). */ -}} {{- $title := partial "tufte/md-title.html" .Title -}} {{- $titleMark := "" -}} {{- if and $title (findRE `HAHAHUGOSHORTCODE|TUFTENEST[0-9a-f]{16}END` $title 1) -}} {{- $tkey := substr (md5 (printf "%s|%s|%s|%s" .Page.RelPermalink .Destination $title (string .Text))) 0 12 -}} {{- $titleMark = printf `%s` $tkey (htmlEscape $title) -}} {{- $title = printf "TUFTEATTR%sEND" $tkey -}} {{- end -}} {{- $text := string .Text -}} {{- $img := "" -}} {{- if or (strings.Contains $text "%s` (htmlEscape $h) (cond (ne $title "") (printf ` title="%s"` (htmlEscape $title)) "") $img -}} {{- printf `%s%s` (replace $text $img $a 1) $titleMark | safeHTML -}} {{- else -}} {{- /* A link whose text is its own address (an autolink , a linkified bare URL, [https://x](https://x/)) gets class "url", like \url: print then does not repeat the address after it. Compared without scheme, "www." and trailing slash, case-insensitively. */ -}} {{- $isURL := false -}} {{- /* An autolink (, , a linkified bare URL or www. address): Hugo passes its label as raw source text, not HTML, which must be escaped here ("?x=1®=2" would otherwise show "®=2"). Recognised by its text being its own destination (Goldmark adds "mailto:" or "http(s)://" to a bare address). A written link whose rendered text equals its destination holds no markup or entity, so escaping it changes nothing. */ -}} {{- $raw := string .Text -}} {{- $dest := string .Destination -}} {{- $auto := and $raw (in (slice $raw (print "mailto:" $raw) (print "http://" $raw) (print "https://" $raw)) $dest) -}} {{- $scheme := `(?i)^(?:[a-z][a-z0-9+.-]*://|mailto:)?(?:www\.)?` -}} {{- $pt := htmlUnescape (strings.TrimSpace (string .PlainText)) | replaceRE $scheme "" | strings.TrimRight "/" | lower -}} {{- $dt := strings.TrimSpace (string .Destination) | replaceRE $scheme "" | strings.TrimRight "/" | lower -}} {{- if or $auto (and $pt (eq $pt $dt)) -}} {{- $isURL = true -}} {{- end -}} {{- /* Dangerous schemes (javascript:, …) are left to html/template, which neutralises them. */ -}} {{ if $auto }}{{ $raw }}{{ else }}{{ .Text }}{{ end }}{{ with $titleMark }}{{ . | safeHTML }}{{ end }} {{- end -}} {{- /* no trailing newline */ -}}
