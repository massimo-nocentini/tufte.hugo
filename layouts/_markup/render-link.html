{{- /* Link render hook. Tufte CSS styles plain elements, so the markup is unchanged; destinations are resolved like Hugo's embedded link hook: a relative destination that names a page (e.g. "../other-post.md" or "/about") or a page/global resource is replaced by its relative permalink, keeping any query and fragment. Fragment-only links ("#x"), absolute URLs and unresolved paths are left as written (in a summary on a list page, tufte/postprocess.html points "#x" to the page: PAGE#x). Notes in the link text ([text[^1]](url), a sidenote, marginnote or cite shortcode) are moved after the by tufte/postprocess.html (tufte/unnest-links.html), and links nested in it are unlinked. A linked stand-alone image, [![alt](src "Caption")](url), reaches the image hook with .IsBlock = true and becomes a figure, which may not sit inside an (nor inside the

Goldmark keeps around the link). The link then goes inside the figure, around the only, and the figure is wrapped in inert block markers:

func/content-raw.html drops the paragraph around a block that is alone in it (the figure stands on its own, like an unlinked block image), and turns a block that shares its paragraph with text into a linked plain image followed by its caption (func/inline-figures.html, which also handles blocks in notes). */ -}} {{- $href := .Destination -}} {{- $u := "" -}} {{- with try (urls.Parse .Destination) -}} {{- if not .Err -}}{{- $u = .Value -}}{{- end -}} {{- end -}} {{- $safe := true -}} {{- with $u -}} {{- if in (slice "javascript" "vbscript" "data" "file") (lower .Scheme) -}} {{- $safe = false -}} {{- end -}} {{- if and .Path (not .IsAbs) (not .Host) -}} {{- $path := strings.TrimPrefix "./" .Path -}} {{- $target := "" -}} {{- with $.PageInner.GetPage $path -}} {{- $target = .RelPermalink -}} {{- else -}} {{- with $.PageInner.Resources.Get $path -}} {{- $target = .RelPermalink -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $path) -}} {{- $target = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- with $target -}} {{- $href = . -}} {{- with $u.RawQuery -}}{{- $href = printf "%s?%s" $href . -}}{{- end -}} {{- with $u.Fragment -}}{{- $href = printf "%s#%s" $href . -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $safe = false -}} {{- end -}} {{- $text := string .Text -}} {{- $img := "" -}} {{- if or (strings.Contains $text "%s` (htmlEscape $h) (cond (ne (printf "%v" (.Title | default "")) "") (printf ` title="%s"` (htmlEscape (printf "%v" .Title))) "") $img -}} {{- printf `%s` (replace $text $img $a 1) | safeHTML -}} {{- else -}} {{- /* A link whose text is its own address (an autolink , a linkified bare URL, [https://x](https://x/)) gets class "url", like \url: print then does not repeat the address after it. Compared without scheme, "www." and trailing slash, case-insensitively. */ -}} {{- $isURL := false -}} {{- $scheme := `(?i)^(?:[a-z][a-z0-9+.-]*://|mailto:)?(?:www\.)?` -}} {{- $pt := htmlUnescape (strings.TrimSpace (string .PlainText)) | replaceRE $scheme "" | strings.TrimRight "/" | lower -}} {{- $dt := strings.TrimSpace (string .Destination) | replaceRE $scheme "" | strings.TrimRight "/" | lower -}} {{- if and $pt (eq $pt $dt) -}} {{- $isURL = true -}} {{- end -}} {{- /* Dangerous schemes (javascript:, …) are left to html/template, which neutralises them. */ -}} {{ .Text }} {{- end -}} {{- /* no trailing newline */ -}}
